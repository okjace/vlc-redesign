# VLC Redesign setup wizard (Linux AppImage). Builds with Qt 6 or Qt 5.
cmake_minimum_required(VERSION 3.16)
project(vlc-redesign-setup LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets)
add_executable(vlc-redesign-setup setup.cpp)
target_link_libraries(vlc-redesign-setup PRIVATE Qt${QT_VERSION_MAJOR}::Widgets)
install(TARGETS vlc-redesign-setup RUNTIME DESTINATION bin)
